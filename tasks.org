This notebook owns the machine-local llama.cpp workflow used by the lab.

Local runtime identity is intentionally literal:

- benchmark id: =qwen3-8b-q4-k-m=
- provider model: =Qwen3-8B-Q4_K_M=
- GGUF: =~/.local/share/models/Qwen3-8B-Q4_K_M.gguf=
- endpoint: =http://127.0.0.1:8000/v1=
- backend: llama.cpp Vulkan on the AMD Radeon RX 5500 XT

Do not replace the provider model with a generic alias such as =local-model=.
The benchmark should record the model that was actually served.

* Phase A — Launch llama.cpp

The launch block is idempotent only when port 8000 is already serving the
expected model. If another model is present it fails loudly rather than
quietly benchmarking the wrong weights. Humanity has enough mislabeled
benchmarks already.

#+name: launch-server
#+begin_src shell :async yes
set -euo pipefail
PORT=8000
MODEL="$HOME/.local/share/models/Qwen3-8B-Q4_K_M.gguf"
PROVIDER_MODEL='Qwen3-8B-Q4_K_M'
LOG=/tmp/llama-server.lab.log
MODELS_JSON=/tmp/rlm-lab-models.json

for pkg in llama-cpp ggml ggml-cpu ggml-vulkan; do
    if ! pacman -Qi "$pkg" >/dev/null 2>&1; then
        echo "MISSING: $pkg"
        echo "install with: sudo pacman -S --needed llama-cpp ggml ggml-cpu ggml-vulkan"
        exit 1
    fi
done

if [ ! -r "$MODEL" ]; then
    echo "model not found: $MODEL"
    exit 1
fi

if curl -fsS "http://127.0.0.1:${PORT}/v1/models" >"$MODELS_JSON" 2>/dev/null; then
    if python3 - "$PROVIDER_MODEL" "$MODELS_JSON" <<'PY'
import json, sys
wanted, path = sys.argv[1], sys.argv[2]
with open(path, encoding="utf-8") as handle:
    data = json.load(handle)
ids = [row.get("id") for row in data.get("data", [])]
print("advertised:", ", ".join(str(x) for x in ids))
raise SystemExit(0 if wanted in ids else 1)
PY
    then
        echo "llama-server already ready on 127.0.0.1:${PORT} with ${PROVIDER_MODEL}"
        exit 0
    fi
    echo "port ${PORT} is serving a different model; stop it before launching this benchmark model"
    exit 1
fi

nohup llama-server \
    --model "$MODEL" \
    --alias "$PROVIDER_MODEL" \
    --host 127.0.0.1 \
    --port "$PORT" \
    --n-gpu-layers 99 \
    --ctx-size 8192 \
    --jinja \
    >"$LOG" 2>&1 </dev/null &

echo "launched llama-server pid=$!"
echo "model=$PROVIDER_MODEL"
echo "endpoint=http://127.0.0.1:${PORT}/v1"
echo "log=$LOG"
#+end_src

* Phase B — Wait for readiness and verify identity

#+name: wait-server
#+begin_src shell :async yes
set -euo pipefail
PORT=8000
PROVIDER_MODEL='Qwen3-8B-Q4_K_M'
MODELS_JSON=/tmp/rlm-lab-models.json

for i in $(seq 1 60); do
    if curl -fsS "http://127.0.0.1:${PORT}/health" 2>/dev/null | grep -q '"status":"ok"'; then
        curl -fsS "http://127.0.0.1:${PORT}/v1/models" >"$MODELS_JSON"
        python3 - "$PROVIDER_MODEL" "$MODELS_JSON" <<'PY'
import json, sys
wanted, path = sys.argv[1], sys.argv[2]
with open(path, encoding="utf-8") as handle:
    data = json.load(handle)
ids = [row.get("id") for row in data.get("data", [])]
print("models:", ", ".join(str(x) for x in ids))
if wanted not in ids:
    raise SystemExit(f"expected {wanted!r}, endpoint advertises {ids!r}")
PY
        echo "ready after ${i}s"
        exit 0
    fi
    sleep 1
done

echo "server not ready after 60s"
tail -60 /tmp/llama-server.lab.log
exit 1
#+end_src

* Phase I — OpenRouter key

This uses Emacs =read-passwd= so the key is not echoed into Org results or
shell history. It updates =.env= in place and sets mode 0600.

#+name: set-openrouter-key
#+begin_src emacs-lisp :results output
(require 'subr-x)
(let* ((root (file-name-directory (or buffer-file-name default-directory)))
       (env-file (expand-file-name ".env" root))
       (template (expand-file-name ".env.example" root))
       (key (string-trim (read-passwd "OpenRouter API key: "))))
  (when (string-empty-p key)
    (error "empty OpenRouter key"))
  (unless (file-exists-p env-file)
    (copy-file template env-file))
  (with-temp-buffer
    (insert-file-contents env-file)
    (goto-char (point-min))
    (if (re-search-forward "^OPENROUTER_API_KEY=.*$" nil t)
        (replace-match (concat "OPENROUTER_API_KEY=" key) t t)
      (goto-char (point-max))
      (unless (bolp) (insert "\n"))
      (insert "OPENROUTER_API_KEY=" key "\n"))
    (goto-char (point-min))
    (if (re-search-forward "^OPENROUTER_BASE_URL=.*$" nil t)
        (replace-match "OPENROUTER_BASE_URL=https://openrouter.ai/api/v1" t t)
      (goto-char (point-max))
      (insert "OPENROUTER_BASE_URL=https://openrouter.ai/api/v1\n"))
    (write-region (point-min) (point-max) env-file nil 'silent))
  (set-file-modes env-file #o600)
  (princ "OpenRouter configuration written; key not displayed\n"))
#+end_src

* Notes

- Shell blocks are asynchronous when =ob-async= is installed.
- The local server binds only to =127.0.0.1:8000=.
- =Qwen3-8B-Q4_K_M= is both the llama.cpp alias and the provider model id.
- =qwen3-8b-q4-k-m= is the benchmark registry id.
- Size-class candidates use their own true model names when loaded; do not
  remap them to the currently loaded Qwen3 8B model.
- Raw rows preserve =provider_model= and =provider_base_url= for auditability.
